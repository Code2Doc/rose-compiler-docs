49template<class Node>
50DOTSynthesizedAttribute
51DOTGeneration<Node>::evaluateSynthesizedAttribute(Node node, DOTInheritedAttribute ia, SubTreeSynthesizedAttributes l)
52 {
53 SubTreeSynthesizedAttributes::iterator iter;
54 ROSE_ASSERT(node);
55 std::string nodeoption;
56 std::string nodelabel = std::string("\\n") + typeid(*node).name();
57 nodelabel += additionalNodeInfo(node);
58 // DQ (11/1/2003) added mechanism to add options (to add color, etc.)
59 nodeoption += additionalNodeOptions(node);
60 switch(traversal)
61 {
62 case TOPDOWNBOTTOMUP:
63 dotrep.addNode(node,dotrep.traceFormat(ia.tdbuTracePos,tdbuTrace)+nodelabel,nodeoption);
64 break;
65 case PREORDER:
66 case TOPDOWN:
67 dotrep.addNode(node,dotrep.traceFormat(ia.tdTracePos)+nodelabel,nodeoption);
68 break;
69 case POSTORDER:
70 case BOTTOMUP:
71 dotrep.addNode(node,dotrep.traceFormat(buTrace)+nodelabel,nodeoption);
72 break;
73 default:
74 assert(false);
75 }
76
77 ++tdbuTrace;
78 ++buTrace;
79 DOTSynthesizedAttribute d(0);
80
81 // add edges or null values
82 for(iter=l.begin();iter!=l.end();iter++)
83 {
84 // DQ (3/5/2007): Added addional function to add more edge information to the label and option string
85 std::string edgelabel = "";
86 std::string edgeoption = "";
87
88 // Use the label to identify the edge (in general, even though in this case the label is "")
89 edgeoption += additionalEdgeOptions(node,iter->node,edgelabel);
90 edgelabel += additionalEdgeInfo (node,iter->node,edgelabel);
91
92 if ( (*iter).node == NULL )
93 {
94 dotrep.addNullValue(node,"",edgelabel,"");
95 }
96 else
97 {
98 switch(traversal)
99 {
100 case TOPDOWNBOTTOMUP:
101 dotrep.addEdge(node,edgelabel,(*iter).node,edgeoption + "dir=both");
102 break;
103 case PREORDER:
104 case TOPDOWN:
105 dotrep.addEdge(node,edgelabel,(*iter).node,edgeoption + "dir=forward");
106 break;
107 case POSTORDER:
108 case BOTTOMUP:
109 dotrep.addEdge(node,edgelabel,(*iter).node,edgeoption + "dir=back");
110 break;
111 default:
112 assert(false);
113 }
114 }
115 }
116
117 d.node=node;
118 ROSE_ASSERT(node!=0);
119 return d;
120 }
