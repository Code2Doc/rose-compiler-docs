912template<class G>
913void
914BaseEmitter<G>::emit(std::ostream &out) const {
915 VMap vmap; // GraphViz node ID for each graph vertex (modified by emit)
916
917 out <<"digraph CFG {\n";
918 out <<" graph [ " <<toString(defaultGraphAttributes_) <<" ];\n";
919 out <<" node [ " <<toString(defaultNodeAttributes_) <<" ];\n";
920 out <<" edge [ " <<toString(defaultEdgeAttributes_) <<" ];\n";
921
922 typedef std::map<std::string /*subgraph name*/, std::string/*subgraph content*/> Subgraphs;
923 Subgraphs subgraphs;
924
925 // Emit vertices to subgraphs
926 for (typename G::ConstVertexIterator vertex=graph_.vertices().begin(); vertex!=graph_.vertices().end(); ++vertex) {
927 const Organization &org = vertexOrganization(vertex);
928 if (org.isSelected() && !vmap.exists(vertex->id())) {
929 std::ostringstream ss;
930 size_t gvid = emitVertex(ss, vertex, org, vmap);
931 vmap.insert(vertex->id(), gvid);
932 subgraphs[org.subgraph()] += ss.str();
933 }
934 }
935
936 // Emit edges to subgraphs
937 for (typename G::ConstEdgeIterator edge=graph_.edges().begin(); edge!=graph_.edges().end(); ++edge) {
938 const Organization &org = edgeOrganization(edge);
939 if (org.isSelected() &&
940 vertexOrganization(edge->source()).isSelected() && vertexOrganization(edge->target()).isSelected()) {
941 std::ostringstream ss;
942 emitEdge(ss, edge, org, vmap);
943 subgraphs[org.subgraph()] += ss.str();
944 }
945 }
946
947 // Emit named subgraphs to output
948 for (const Subgraphs::value_type &node: subgraphs) {
949 const std::string &subgraphName = node.first;
950 const std::string &subgraphContent = node.second;
951 if (!subgraphName.empty()) {
952 out <<"\nsubgraph cluster_" <<subgraphName <<" {"
953 <<" label=" <<subgraphOrganization(subgraphName).label() <<" "
954 <<toString(subgraphOrganization(subgraphName).attributes()) <<"\n"
955 <<subgraphContent
956 <<"}\n";
957 }
958 }
959
960 // Emit unnamed subgraph content without a surrounding subgraph construct (i.e., global graph)
961 Subgraphs::iterator unnamedSubgraph = subgraphs.find("");
962 if (unnamedSubgraph != subgraphs.end())
963 out <<unnamedSubgraph->second;
964
965 // Emit pseudo edges
966 for (const PseudoEdge &edge: pseudoEdges_) {
967 if (vertexOrganization(edge.src).isSelected() && vertexOrganization(edge.dst).isSelected()) {
968 std::string sourceName = vertexOrganization(edge.src).name();
969 std::string targetName = vertexOrganization(edge.dst).name();
970 out <<(sourceName.empty() ? StringUtility::numberToString(vmap[edge.src->id()]) : sourceName)
971 <<" -> "
972 <<(targetName.empty() ? StringUtility::numberToString(vmap[edge.dst->id()]) : targetName)
973 <<" [ label=" <<escape(edge.label) <<" ];\n";
974 }
975 }
