88 assert(nbits>0 && nbits<=8*sizeof value);
89 uint64_t sign_bit = ((uint64_t)1 << (nbits-1));
90
91 // Hexadecimal value
92 std::string retval;
93 int nnibbles = (nbits+3)/4;
94 std::string fmt = "0x%0" + boost::lexical_cast<std::string>(nnibbles) + "x";
95 retval = (boost::format(fmt) % value).str();
96
97 // unsigned decimal
98 bool showed_unsigned = false;
99 if (show_unsigned_decimal && value >= decimal_threshold) {
100 retval = appendAsmComment(retval, numberToString(value));
101 showed_unsigned = true;
102 }
103
104 // signed decimal
105 if (show_signed_decimal) {
106 if (0 == (value & sign_bit)) {
107 // This is a positive value. Don't show it if we did already.
108 if (!showed_unsigned && value >= decimal_threshold)
109 retval = appendAsmComment(retval, numberToString(value));
110 } else {
111 // This is a negative value, so show it as negative. We have to manually sign extend it first.
112 int64_t signed_value = (value | (-1ll & ~(sign_bit-1)));
113 retval = appendAsmComment(retval, numberToString(signed_value));
114 }
