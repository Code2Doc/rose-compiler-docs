164// Escaping special characters in shells is difficult. There are many classes of characters:
165//
166// 1. Characters that can appear unescaped and outside quotes (bare). They can also appear unescaped inside single or
167// double quotes. Most characters fall into this category.
168//
169// 2. Characters that must be escaped when outside quotes or (not escaped) inside double or single quotes. These are
170// characters like "*", "?", "=", "{", "}", "[", "]", etc. and consist of most printable non-alphanumeric characters.
171//
172// 3. Characters that cannot be escaped by some shells. These are things like non-printable control characters
173// and characters above 0177. The Bash shell has a special syntax for escaping these in a C-like manner.
174std::string
175bourneEscape(const std::string &s) {
176 // If the string is empty it needs to be quoted.
177 if (s.empty())
178 return "''";
179
180 // The presence of non-printing characters or single quotes trumps all others and requires C-style quoting
181 for (char ch: s) {
182 if (!::isprint(ch) || '\'' == ch)
183 return "$'" + cEscape(s, '\'') + "'";
184 }
185
186 // If the string contains any shell meta characters or white space that must be quoted then single-quote the entire string
187 // and escape backslashes.
188 for (char ch: s) {
189 if (!::isalnum(ch) && !strchr("_-+./=", ch))
190 return "'" + boost::replace_all_copy(s, "\\", "\\\\") + "'";
191 }
192
193 // No quoting or escaping necessary
194 return s;
195}
