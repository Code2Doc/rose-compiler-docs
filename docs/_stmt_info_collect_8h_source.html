173/* Support tracking modification of variables, by building a map from each variable name
174 * to all the AST nodes where the variable is modified */
175template <class Select>
176class ModifyVariableMap : public StmtSideEffectCollect
177{
178 class VarModSet : public std::set<AstNodePtr> {};
179 typedef std::map <std::string, VarModSet, std::less<std::string> > VarModInfo;
180 VarModInfo varmodInfo;
181 Select sel;
182 std::function<bool(AstNodePtr,AstNodePtr)> collect_modify;
183
184 public:
185 AstInterface& get_astInterface() { return fa_; }
186 ModifyVariableMap(AstInterface& _fa, Select _sel,
187 FunctionSideEffectInterface* a=0)
188 : StmtSideEffectCollect(_fa,a), sel(_sel) {
189 collect_modify = [this](AstNodePtr mod_first, AstNodePtr /*mod_second*/) {
190 std::string varname;
191 if (fa_.IsVarRef(mod_first,0, &varname)) {
192 AstNodePtr l = fa_.GetParent(mod_first);
193 VarModSet& cur = varmodInfo[varname];
194 for ( ; l != 0; l = fa_.GetParent(l)) {
195 if (sel(fa_,l))
196 cur.insert(l);
197 }
198 }
199 return true;
200 };
201 StmtSideEffectCollect::set_modify_collect(collect_modify);
202 }
203 void Collect(const AstNodePtr& root)
204 {
205 StmtSideEffectCollect::get_side_effect(fa_, root);
206 }
207 bool Modify( const AstNodePtr& l, const std::string& varname) const
208 {
209 typename VarModInfo::const_iterator p = varmodInfo.find(varname);
210 if (p != varmodInfo.end()) {
211 if (l == 0)
212 return true;
213 const VarModSet& cur = (*p).second;
214 return cur.find(l) != cur.end();
215 }
216 else
217 return false;
218 }
219};
