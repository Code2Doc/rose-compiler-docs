63template <class T>
64T escape_double_quotes(const T & input) {
65 std::string result(input);
66 size_t cur = result.find("\"");
67 while (cur < result.size() && cur != std::string::npos) {
68 result.replace(cur, 1, "\\\"");
69 cur = result.find("\"", cur+2);
70 }
71 return result;
72}
