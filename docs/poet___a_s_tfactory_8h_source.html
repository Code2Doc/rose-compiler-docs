37class MultiFactory

39 typedef std:: map<POETCode*, POETCode*> LastMap;
40 LastMap lastMap;
41 typedef std:: map<POETCode*, LastMap*> FirstMap;
42 FirstMap firstMap;
43 virtual POETCode* new_code(POETCode* first, POETCode* rest) = 0;
44 public:
45 POETCode* new_item(POETCode* first, POETCode* rest) {
46 LastMap* m = 0;
47 FirstMap::const_iterator p = firstMap.find(first);
48 if (p == firstMap.end()) {
49 firstMap[first] = m = new LastMap();
50 }
51 else
52 m = (*p).second;
53 LastMap::const_iterator p2 = m->find(rest);
54 if (p2 == m->end()) {
55 return (*m)[rest] = new_code(first, rest);
56 }
57 else
58 return (*p2).second;
59 }
