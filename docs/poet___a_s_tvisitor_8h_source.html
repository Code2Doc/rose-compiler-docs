114class ReplInfoVisitor : public EvaluatePOET, public POETCodeVisitor
115{
116 protected:
117 POETCode* res;
118 std::set<LocalVar*> modvars;
119 public:
120 ReplInfoVisitor() { res = 0; }
121 virtual void defaultVisit(POETCode* c) = 0;
122 virtual void visitLocalVar(LocalVar* v);
123 virtual void visitAssign(POETAssign* assign) {
124 POETCode* lhs = apply(assign->get_lhs());
125 POETCode* rhs = apply(assign->get_rhs());
126 if (lhs == assign->get_lhs() && rhs == assign->get_rhs())
127 res = assign;
128 else
129 res = new POETAssign(lhs,rhs);
130 }
131 virtual void visitList(POETList* l)
132 {
133 POETCode* first = l->get_first(), *rest = l->get_rest();
134 apply(first);
135 POETCode* r1 = res;
136 if (r1 == 0) {
137 if (rest == 0) res = 0;
138 else apply(rest);
139 }
140 else if (rest != 0) {
141 apply(rest);
142
143 POETCode* r2 = res;
144 res = MakeXformList(r1,r2,false);
145 if (res == 0) res = fac->new_list(r1,r2);
146 else res->visit(this);
147 }
148 else res = fac->new_list(r1,0);
149 }
150 virtual void visitTuple(POETTuple* v)
151 {
152 int size = v->size();
153 std::vector<POETCode*> cur(size);
154 bool change=false;
155 for (int i = 0; i < size; ++i) {
156 POETCode* elem1 = v->get_entry(i);
157 POETCode* elem2 = apply(elem1);
158 if (elem2 != elem1) change = true;
159 if (elem2 == 0) elem2 = EMPTY_LIST;
160 cur[i] = elem2;
161 }
162 if (!change) { res = v; return;}
163 assert(size >= 2);
164 res = 0;
165 for (int i = 0; i < size; ++i)
166 res = fac->append_tuple(res, cur[i]);
167 }
168 virtual void visitCodeVar( CodeVar* v)
169 {
170 POETCode *nargs = (v->get_args() == 0)? 0 : apply(v->get_args());
171/*
172 POETCode* nattr = (v->get_attr() == 0)? (POETCode*)0 : apply(v->get_attr());
173*/
174 if (nargs != v->get_args()) // || nattr != v->get_attr())
175 res = ASTFactory::inst()->build_codeRef(v->get_entry(), nargs, v->get_attr());
176 else res = v;
177 }
178 virtual void visitUnknown(POETCode_ext* ext)
179 {
180 POETCode* res1 = POETAstInterface::visitAstChildren(ext->get_content(),this);
181 if (res1 != 0) res = res1;
182 else res = ext;
183 }
184 virtual POETCode* apply(POETCode* code)
185 { res = code; code->visit(this); return res; }
186};

188inline void ReplInfoVisitor:: visitLocalVar(LocalVar* v) {
189 if (modvars.find(v) != modvars.end()) {
190 if (user_debug)
191 std::cerr << "LocalVar visited more than--so skip:" << v->toString() << "\n";
192 res = v; return;
193 }
194 LvarSymbolTable::Entry e = v->get_entry();
195 POETCode* save = e.get_code();
196 if (save == 0) {
197 if (e.get_entry_type() == LVAR_TRACE) TRACE_RECURSIVE(v)
198 else { res = v; return; }
199 }
200 if (save != v) {
201 e.set_code(0);
202 res = apply(save);
203 if (v != res) {
204 if (user_debug)
205 std::cerr << "setting LocalVar " << v->toString() << " with " << res->toString() << "\n";
206 e.set_code(res);
207 modvars.insert(v);
208 if (e.get_entry_type() == LVAR_TRACE)
209 res = v;
210 }
211 else if (e.get_code() == 0) {
212 if (user_debug)
213 std::cerr << "LocalVar unchanged: " << v->toString() << "\n";
214 e.set_code(save);
215 }
216 }
217 }
