184inline bool isPowerOfTwo(T value)
185{
186 assert(sizeof(T) <= sizeof(uint64_t));
187 if (0 != (value & SHL1<T, 8*sizeof(T)-1>::value))
188 value = ~value + 1;
189 uint64_t uval = value;
190 while (uval) {
191 if (uval & 1)
192 return 1==uval;
193 uval >>= 1u;
194 }
