67// Check if an expression is an member access/member function call of a known array type.
68template <class Descriptor>
69std::string CPPTypeCollection<Descriptor>::
70is_known_member_function( CPPAstInterface& fa, const AstNodePtr& exp,
71 AstNodePtr* objp, AstInterface::AstNodeList* args,
72 Descriptor* desc)
73{
74 AstNodePtr obj;
75 std::string fname;
76 if (!fa.IsMemberAccess( exp, &obj, &fname) &&
77 !fa.IsMemberFunctionCall(exp, &obj, &fname, 0, args))
78 return "";
79 if (obj == AST_NULL)
80 return "";
81 if (cp->known_type( fa, obj, desc)) {
82 if (objp != 0)
83 *objp = obj;
84 return fname;
85 }
86 return "";
87}
