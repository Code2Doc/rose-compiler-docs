83inline POETString* AST2String(POETCode* r)
84{
85 switch (r->get_enum()) {
86 case SRC_STRING: return static_cast<POETString*>(r);
87 case SRC_ICONST: return ASTFactory::inst()->new_string(r->toString());
88 case SRC_LIST:
89 {
90 for (POETList* cur = static_cast<POETList*>(r);
91 cur != 0; cur = cur->get_rest())
92 {
93 POETString* cur_string = AST2String(cur->get_first());
94 if (cur_string == 0) return 0;
95 }
96 return ASTFactory::inst()->new_string(r->toString(OUTPUT_NO_DEBUG));
97 }
98 default: return 0;
99 }
