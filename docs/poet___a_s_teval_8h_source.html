81inline POETString* AST2String(POETCode* r)
82{
83 if (r == EMPTY_LIST) return 0;
84 switch (r->get_enum()) {
85 case SRC_STRING: return static_cast<POETString*>(r);
86 case SRC_ICONST: return ASTFactory::inst()->new_string(r->toString());
87 case SRC_LIST:
88 {
89 for (POETList* cur = static_cast<POETList*>(r);
90 cur != 0; cur = cur->get_rest())
91 {
92 POETString* cur_string = AST2String(cur->get_first());
93 if (cur_string == 0) return 0;
94 }
95 return ASTFactory::inst()->new_string(r->toString(OUTPUT_NO_DEBUG));
96 }
97 case SRC_UNKNOWN: {
98 std::string s = POETAstInterface::Ast2String(static_cast<POETCode_ext*>(r)->get_content());
99 if (s != "") { return ASTFactory::inst()->new_string(s); }
100 else return 0;
101 }
102 default: return 0;
103 }
104}

106inline POETIconst* AST2Iconst(POETCode* code)
107{

112 code = l->get_first(); t = code->get_enum();

115 case SRC_ICONST: return static_cast<POETIconst*>(code);
116 case SRC_STRING: {
