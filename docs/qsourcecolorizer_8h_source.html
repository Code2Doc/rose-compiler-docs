117 //kdDebug(9032) << "KeywordsHLItem::checkHLItem" << endl;
118
119 int start_pos = pos;
120
121 while( (pos<length) && (buffer[pos].isLetterOrNumber() || buffer[pos] == '_') )
122 ++pos;
123
124 if( start_pos != pos ) {
125 if ( m_ignoreCase ) {
126 m_ok = m_keywords.contains( QString(buffer+start_pos, pos-start_pos).lower() );
127 } else {
128 m_ok = m_keywords.contains( QString(buffer+start_pos, pos-start_pos) );
129 }
130 }
131
132 return ( m_ok || m_finalize ) ? pos : start_pos;
133 }
