284 Rose_STL_Container <T*>
285 queryNodeList ( Rose_STL_Container <T*> queryList, VariantVector targetVector)
286 {
287 Rose_STL_Container <T*> result;
288 // Rose_STL_Container <T*>::iterator iter; // not recognized ??
289 size_t i=0;
290 for (i = 0; i < queryList.size(); i++)
291 {
292 SgNode* node = dynamic_cast<SgNode*> (queryList[i]);
293 ROSE_ASSERT(node != NULL);
294 for (VariantVector::const_iterator iter = targetVector.begin(); iter != targetVector.end(); iter++)
295 {
296 if (node->variantT() == *iter)
297 {
298 result.push_back(queryList[i]);
299 break;
300 }
301 }
302 }
303
304 return result;
305 }
