329 AstInterface::TraversalVisitType t)

353 }

361 if (trueNode != 0)
362 AddBranch( trueNode, exitNode, CFGConfig::ALWAYS);

365 }
366 if (falseNode != 0)

404 virtual bool ProcessGoto( AstInterface &fa, const AstNodePtr& s,
405 const AstNodePtr& _dest)
406 {
407 if (debug_cfg()) {
408 std::cerr << "processing go to " ;
409 }
410 AstNodePtr dest = _dest;
411 Node *lastNode = GetCurNode( true );
412 AddStmt(fa, lastNode, s);
413
414 if (fa.IsGotoAfter(s)) {
415 Node* destNode = MapStmt(fa, dest, EXIT);
416 AddBranch( lastNode, destNode, CFGConfig::ALWAYS);
417 AstNodeList blockstmts;
418 while (fa.IsBlock(dest, 0, &blockstmts)) {
419 dest = blockstmts.back();
420 SetStmtNode(dest, destNode, EXIT);
421 }
422 }
423 else {
424 Node* destNode = MapStmt(fa, dest, START);
425 AddBranch( lastNode, destNode, CFGConfig::ALWAYS);
426 AstNodePtr prev = fa.GetPrevStmt(dest);
427 if (prev != AST_NULL)
428 SetStmtNode( prev, destNode, EXIT);
429 }
430 SetCurNode(0);
431
432 return ProcessAstTree::ProcessGoto(fa, s, dest);

435 virtual bool ProcessStmt(AstInterface &fa, const AstNodePtr& s)
436 {
437 if (debug_cfg())
438 std::cerr << "processing stmt " << AstInterface::AstToString(s) << std::endl;
439
440 Node *lastNode = GetCurNode();
441 Node *n = GetStmtNode(s, START);

443 AddStmt(fa, lastNode, s);

445 AddStmt(fa, lastNode, s);

447 }

457 }
