258 template <class T>
259 bool
260 NodeCollectorByLoc<T>::matchesLoc (SgLocatedNode* node) const
261 {
262 if (node == NULL)
263 return false;
264
265 Sg_File_Info* info_start = node->get_startOfConstruct ();
266 ROSE_ASSERT (info_start != NULL);
267
268 if (GenUtil::isSamePathname (filename_, info_start->get_raw_filename ()))
269 return false;
270 // else, at least source file name matches
271
272 int line_start = info_start->get_line ();
273 Sg_File_Info* info_end = node->get_endOfConstruct ();
274 int line_end = (info_end == NULL) ? line_start : info_end->get_line ();
275
276 if (line_end < line_start)
277 line_end = line_start;
278
279 return linesMatch (line_start, line_end);
280 }
