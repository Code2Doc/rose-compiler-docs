66 performTraverse( SgNode *astNode, InheritedAttribute inheritedValue )

68 // check for null pointer and if we are in the file
69 if( astNode && inFileToTraverse( astNode, traversalConstraint, fileToVisit ) )
70 {
71 const size_t numSuccessors( astNode->get_numberOfTraversalSuccessors() );
72 // Pre evaluation ( aka inherited Attribute )
73 inheritedValue = evaluateInheritedAttribute( astNode, inheritedValue );
74
75 // loop through children ...
76 for( size_t index = 0; index < numSuccessors; ++index )
77 {
78 SgNode *child = NULL;
79 child = astNode->get_traversalSuccessorByIndex( index );
80 // process child
81 synthesizedAttributes->push( performTraverse( child, inheritedValue ) );
82 }
83 synthesizedAttributes->setFrameSize( numSuccessors );
84 ROSE_ASSERT( synthesizedAttributes->size() == numSuccessors );
85 // Post evaluation ( aka synthesizedAttributes )
86 return evaluateSynthesizedAttribute( astNode, inheritedValue, *synthesizedAttributes );
87 } // end if( astNode && inFileToTraverse( astNode, traversalConstraint, fileToVisit ) )
