263 BiDirNodesIterator (BaseGraph&) { dir = Forward; iter = gr->node_set.begin(); }
264 BiDirNodesIterator (BaseGraph& g, dirType d) {
265 gr = &g; dir = d;
266 if (dir == Forward)
267 iter = gr->node_set.begin();
268 else {
269 iter = gr->node_set.end();
270 iter--;
271 pre_begin = gr->node_set.begin();
272 --pre_begin;
273 }
274 }
275 ~BiDirNodesIterator () {}
276 void operator++ () { ++iter; }
277 void operator-- () { --iter; }
278 operator bool () {
279 if (dir == Forward)
280 return iter != gr->node_set.end();
281 else {
282 return iter != pre_begin;
283 }
284 }
