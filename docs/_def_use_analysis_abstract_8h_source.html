58 template <typename T>
59 SgNode* getOtherInNode(T cfgNode, SgNode* oneNode){
60 SgNode* otherNode = NULL;
61 std::vector<filteredCFGEdgeType > in_edges = cfgNode.inEdges();
62 for (std::vector<filteredCFGEdgeType >::const_iterator i = in_edges.begin(); i != in_edges.end(); ++i) {
63 filteredCFGEdgeType filterEdge = *i;
64 T filterNode = filterEdge.source();
65 if (filterNode.getNode()!=oneNode)
66 otherNode = filterNode.getNode();
67 }
68 if (DEBUG_MODE_EXTRA)
69 std::cout << "getOtherInNode:: other: " << otherNode << " previous: " <<
70 oneNode << " size of in: " << in_edges.size() << std::endl;
71 return otherNode;
72 }
