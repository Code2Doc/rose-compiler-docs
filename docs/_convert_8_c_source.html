214encode_base64(const uint8_t *data, size_t nbytes, bool do_pad) {
215 static const char *digits = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
216 std::string s;
217 unsigned val = 0; // only low-order 24 bits used
218 for (size_t i=0; i<nbytes; ++i) {
219 switch (i%3) {
220 case 0:
221 val = IntegerOps::shiftLeft2<unsigned>(data[i], 16);
222 break;
223 case 1:
224 val |= IntegerOps::shiftLeft2<unsigned>(data[i], 8);
225 break;
226 case 2:
227 val |= data[i];
228 s += digits[IntegerOps::shiftRightLogical2(val, 18) & 0x3f];
229 s += digits[IntegerOps::shiftRightLogical2(val, 12) & 0x3f];
230 s += digits[IntegerOps::shiftRightLogical2(val, 6) & 0x3f];
231 s += digits[val & 0x3f];
232 break;
233 }
234 }
235 switch (nbytes % 3) {
236 case 0:
237 break;
238 case 1:
239 s += digits[IntegerOps::shiftRightLogical2(val, 18) & 0x3f];
240 s += digits[IntegerOps::shiftRightLogical2(val, 12) & 0x3f];
241 if (do_pad)
242 s += "==";
243 break;
244 case 2:
245 s += digits[IntegerOps::shiftRightLogical2(val, 18) & 0x3f];
246 s += digits[IntegerOps::shiftRightLogical2(val, 12) & 0x3f];
247 s += digits[IntegerOps::shiftRightLogical2(val, 6) & 0x3f];
248 if (do_pad)
249 s += "=";
250 break;
251 }
252 return s;
253}

255std::vector<uint8_t>
256decode_base64(const std::string &s) {
257 std::vector<uint8_t> retval;
258 retval.reserve((s.size()*3)/4+3);
259 unsigned val=0, ndigits=0;
260 for (size_t i=0; i<s.size(); ++i) {
261 unsigned digit;
262 if (s[i]>='A' && s[i]<='Z') {
263 digit = s[i]-'A';
264 } else if (s[i]>='a' && s[i]<='z') {
265 digit = s[i]-'a' + 26;
266 } else if (s[i]>='0' && s[i]<='9') {
267 digit = s[i]-'0' + 52;
268 } else if (s[i]=='+') {
269 digit = 62;
270 } else if (s[i]=='/') {
271 digit = 63;
272 } else {
273 continue;
274 }
275
276 // only reached for valid base64 characters
277 val |= IntegerOps::shiftLeft2(digit, 18-6*ndigits);
278 if (4==++ndigits) {
279 retval.push_back(IntegerOps::shiftRightLogical2(val, 16) & 0xff);
280 retval.push_back(IntegerOps::shiftRightLogical2(val, 8) & 0xff);
281 retval.push_back(val & 0xff);
282 val = ndigits = 0;
283 }
284 }
285
286 assert(0==ndigits || 2==ndigits || 3==ndigits);
287 if (ndigits>=2) {
288 retval.push_back(IntegerOps::shiftRightLogical2(val, 16) & 0xff);
289 if (ndigits==3)
290 retval.push_back(IntegerOps::shiftRightLogical2(val, 8) & 0xff);
291 }
292 return retval;
293}

297unsigned long
298generate_checksum(std::string s) {
299 std::string uniqueName = s;
300
301 // The checksum function requires a even length string (so we have to fix it up)
302 if (uniqueName.size() % 2 != 0)
303 {
304 // printf ("Adding another character to make string an even valued length \n");
305 // Use a character that does not appear in mangled
306 // names so that no other mangled name could include it.
307 uniqueName += "#";
308 }
309 assert(uniqueName.size() % 2 == 0);
310
311 // Call a simple checksum function
312 // unsigned short int checksum = StringUtility::chksum(buffer,uniqueName.size());
313
314 unsigned long accum = 0;
315 unsigned long chksm;
316
317 accum = 0;
318 unsigned int len = uniqueName.size() / 2; /* Words only */
319 for (unsigned int i=0; i< len; i++)
320 accum += (unsigned long) (((unsigned short*)uniqueName.data())[i]);
321
322 chksm = (accum & 0xffff); /* Mask all but low word */
323 chksm += (accum >> 16); /* Sum all the carries */
324
325 if (chksm > 0xffff) /* If this also carried */
326 chksm++; /* Sum this too */
327 unsigned short checksum = (unsigned short) (chksm & 0xffff);
328
329 // printf ("Checksum = %d \n",checksum);
330 return (unsigned long) checksum;
331}
