49{

51 const ReachingDefinitionGenerator* g = r.get_generator();
52 StmtSideEffectCollect collect(fa, f);
53
54 std::map<AstNodePtr, Node*> defmap;
55 const ReachingDefinitionBase& base = g->get_base();
56 for (ReachingDefinitionBase::iterator p = base.begin(); p != base.end(); ++p) {
57 std::pair<AstNodePtr,AstNodePtr> cur = base.get_ref(p);
58 if (DebugDefUseChain())
59 std::cerr << "creating def node : " << AstInterface::AstToString(cur.first) << " : " << AstInterface::AstToString(cur.second) << std::endl;
60 Node* n = CreateNode( fa, cur.first, cur.second, true);
61 assert(n != 0);
62 defvec.push_back(n);
63 defmap[cur.first] = n;
64 }
65 for (ReachingDefinitionAnalysis::NodeIterator p = r.GetNodeIterator();
66 !p.ReachEnd(); ++p) {
67 ReachingDefNode* cur = *p;
68 if (DebugDefUseChain()) {
69 std::cerr << "processing CFG node : ";
70 cur->write(std::cerr);
71 }
72 ReachingDefinitions in = cur->get_entry_defs();
73 if (DebugDefUseChain()) {
74 std::cerr << "Reaching definitions: \n";
75 DumpDefSet(defvec,in);
76 std::cerr << std::endl;
77 }

79 auto CreateEdges = [&fa,&g,&alias,this,&in,&defvec]
80 (Node* cur, const AstNodePtr& ref) {
81 ReachingDefinitions known = g->get_empty_set();
82 ReachingDefinitions unknown = in;
83 std::string varname;
84 AstNodePtr scope;
85 if (fa.IsVarRef(ref, 0, &varname, &scope)) {
86 unknown = known = g->get_def_set(varname, scope);
87 unknown.complement();
88 known &= in;
89 unknown &= in;
90 }
91 for (size_t i = 0; i < defvec.size(); ++i) {
92 Node* def = defvec[i];
93 assert (def != 0);
94 if (known.has_member(i) ||
95 (unknown.has_member(i) && alias.may_alias( fa, ref, def->get_ref()) )) {
96 if (DebugDefUseChain())
97 std::cerr << " creating edge from " << def->toString() << std::endl;
98 this->CreateEdge(def, cur);
99 }
100 else if (DebugDefUseChain()) {
101 if (!unknown.has_member(i))
102 std::cerr << "not in reaching definition: " << def->toString();
103 else if (!alias.may_alias( fa, ref, def->get_ref()))
104 std::cerr << "not aliased: " << def->toString();
105 }
106 }
107 };
108 // Collecting read set.
109 std::function<bool(AstNodePtr,AstNodePtr)> opread =
110 [&defvec, this, &fa, &in, &CreateEdges]
111 ( AstNodePtr read_first, AstNodePtr read_second) {
112 if (DebugDefUseChain()) {

117 if (cur == 0) {

130 AstNodePtr scope;

139 std::pair<AstNodePtr,AstNodePtr> mod(mod_first, mod_second);
140 if (fa.IsVarRef(mod_first, 0, &varname, &scope)) {
141 g->add_def( in, varname, scope, mod);

145 }

147 std::cerr << "finish processing gen mod info : " << AstInterface::AstToString(mod_first) << " : " << AstInterface::AstToString(mod_second) << std::endl;
148 DumpDefSet(defvec,in);
149 }

151 };

153 [&defvec, g, &fa, &in]
154 (AstNodePtr mod_first, AstNodePtr mod_second) {
155 std::string varname;
156 AstNodePtr scope;
157 if (DebugDefUseChain()) {
158 std::cerr << "processing kill mod info : " << AstInterface::AstToString(mod_first) << " : " << AstInterface::AstToString(mod_second) << std::endl;
159 DumpDefSet(defvec,in);
160 }
161 if (fa.IsVarRef(mod_first, 0, &varname, &scope)) {
162 ReachingDefinitions kill = g->get_def_set(varname, scope);
163 kill.complement();
164 in &= kill;
165 }
166 if (DebugDefUseChain()) {

168 DumpDefSet(defvec,in);
169 }
170 return true;
171 };
172 collect.set_modify_collect(opgen);
173 collect.set_read_collect(opread);
174 collect.set_kill_collect(opkill);
175 std::list <AstNodePtr>& stmts = cur->GetStmts();
176 for (std::list<AstNodePtr>::iterator p = stmts.begin(); p != stmts.end();

178 AstNodePtr cur = *p;

182 }
183 }

185 if (DebugDefUseChain()) {

187 std::cerr << GraphToString(*this);

207template<class Node>
208void PropagateDefUseChainUpdate( DefUseChain<Node> *graph,
209 UpdateDefUseChainNode<Node>& update)
210{
211 std::set<Node*> worklist;
212 AppendWorkListWrap<Node> append(worklist);
213 update.init(append);
214 while (worklist.size()) {
215 Node* cur = *worklist.begin();
216 worklist.erase(worklist.begin());
217 if (cur->is_definition()) {
218 for (GraphNodeSuccessorIterator<DefUseChain<Node> > usep(graph,cur);
219 !usep.ReachEnd(); ++usep) {
220 Node* use = *usep;
221 if (use->is_definition())
222 continue;
223 GraphNodePredecessorIterator<DefUseChain<Node> > defp(graph,use);
224#ifndef NDEBUG
225 Node *tmp = *defp;
226#endif
227 ++defp;
228 if (defp.ReachEnd()) {
229 assert( tmp == cur);
230 update.update_use_node(use, cur, append);
231 }
232 }
233 }
234 else {
235 GraphNodePredecessorIterator<DefUseChain<Node> > defp(graph,cur);
236 if (defp.ReachEnd()) {
237 if (DebugDefUseChain()) {
238 std::cerr << "Error: use of reference with no definition: ";
239 cur->Dump();
240 std::cerr << std::endl;

242 }

244 // Node* def = *defp;
245 if (!defp.ReachEnd()) {
246 Node* def = *defp;

249 if (update.update_def_node(def, cur, append))
