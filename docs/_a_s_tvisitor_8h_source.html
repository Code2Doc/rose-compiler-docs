113 virtual void visitAssign(POETAssign* assign) {
114 POETCode* lhs = apply(assign->get_lhs());
115 POETCode* rhs = apply(assign->get_rhs());
116 if (lhs == assign->get_lhs() && rhs == assign->get_rhs())

121 virtual void visitList(POETList* l)
122 {
123 POETCode* first = l->get_first(), *rest = l->get_rest();
124 apply(first);
125 POETCode* r1 = res;
126 if (r1 == 0) {
127 if (rest == 0) res = 0;
128 else apply(rest);
129 }
130 else if (rest != 0) {
131 bool inList_save = inList;
132
133 inList = true;
134 apply(rest);
135 inList=inList_save;
136
137 POETCode* r2 = res;
138 res = MakeXformList(r1,r2,false);
139 if (res == 0) res = fac->new_list(r1,r2);

141 }
142 else if (inList) res = fac->new_list(r1,0);
143 else res = r1;
144 }
145 virtual void visitTuple(POETTuple* v)
146 {
147 int size = v->size();
148 std::vector<POETCode*> cur(size);
149 bool change=false;
150 for (int i = 0; i < size; ++i) {
151 POETCode* elem1 = v->get_entry(i);
152 POETCode* elem2 = apply(elem1);
153 if (elem2 != elem1) change = true;
154 if (elem2 == 0) elem2 = EMPTY;
155 cur[i] = elem2;
156 }
157 if (!change) { res = v; return;}
158 assert(size >= 2);
159 res = 0;
160 for (int i = 0; i < size; ++i)
161 res = fac->append_tuple(res, cur[i]);
162 }
163 virtual void visitCodeVar( CodeVar* v)
164 {
165 if (v->get_args() != 0) {
166 POETCode *nargs = apply(v->get_args());
167 if (nargs != v->get_args())
168 res = fac->build_codeRef(v, nargs, true);
169 else res = v;
170 }
171 else
172 res = v;
173 }
